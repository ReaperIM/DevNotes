# Поиск значения в массиве, игнорируя регистр
```php
# bySanya@
/*
Описание: Поиск значения в массиве, игнорируя регистр.
Параметры: 
	$needle - значение для поиска
	$haystack - массив, в котором осуществляется поиск
Возвращает:
	- true, если значение найдено
	- false, если значение не найдено
*/
function in_array_strtolower( string $needle, array $haystack ) :bool
{
	$needle = mb_strtolower($needle);
	$strtolower_arr = array_map('mb_strtolower', $haystack);

	return in_array($needle, $strtolower_arr);
}
```

# Транслитерация текста с кириллицы на латиницу
```php
# bySanya@
/*
Описание: Транслитерация текста с кириллицы на латиницу.
Параметры: 
	$text - текст для транслитерации
Возвращает:
	- строку с транслитерированным текстом
*/
function transliterate( string $text ) :string
{
	if( preg_match('/[\p{Cyrillic}]/u', $text) )
	{
		$translit_arr = [
			'А' => 'A', 'Б' => 'B', 'В' => 'V', 'Г' => 'G', 'Д' => 'D', 'Е' => 'E', 'Ё' => 'YO',
			'Ж' => 'ZH', 'З' => 'Z', 'И' => 'I', 'Й' => 'Y', 'К' => 'K', 'Л' => 'L', 'М' => 'M',
			'Н' => 'N', 'О' => 'O', 'П' => 'P', 'Р' => 'R', 'С' => 'S', 'Т' => 'T', 'У' => 'U',
			'Ф' => 'F', 'Х' => 'KH', 'Ц' => 'TS', 'Ч' => 'CH', 'Ш' => 'SH', 'Щ' => 'SHCH', 'Ы' => 'Y',
			'Э' => 'E', 'Ю' => 'YU', 'Я' => 'YA',
			'а' => 'a', 'б' => 'b', 'в' => 'v', 'г' => 'g', 'д' => 'd', 'е' => 'e', 'ё' => 'yo',
			'ж' => 'zh', 'з' => 'z', 'и' => 'i', 'й' => 'y', 'к' => 'k', 'л' => 'l', 'м' => 'm',
			'н' => 'n', 'о' => 'o', 'п' => 'p', 'р' => 'r', 'с' => 's', 'т' => 't', 'у' => 'u',
			'ф' => 'f', 'х' => 'kh', 'ц' => 'ts', 'ч' => 'ch', 'ш' => 'sh', 'щ' => 'shch', 'ы' => 'y',
			'э' => 'e', 'ю' => 'yu', 'я' => 'ya'
		];

		return strtr($text, $translit_arr);
	}
	return $text;
}
```

# Установка статуса страницы ошибки и добавление сообщения об ошибке
```php
# bySanya@
/*
Описание: Установка статуса страницы ошибки и добавление сообщения об ошибке.
Параметры: 
	$code - код ошибки (по умолчанию 404)
	$message - сообщение об ошибке (по умолчанию "Страница не найдена")
Возвращает: отсутствует
*/
function set_error_page_status( int $code = 404, string $message = 'Страница не найдена' )
{
	header('HTTP/1.1 '.$code.' '.$message);
	header('Status: '.$code.' '.$message);
	
	echo '['.$code.'] '.$message;
	
	exit( );
}
```
